Sample SQL Operations & Relationship Logic
-- Retrieve Campaign Funding Status & Total Backers
SELECT 
    c.campaign_id,
    c.title,
    c.goal_amount,
    COUNT(p.pledge_id) AS total_backers,
    COALESCE(SUM(p.pledge_amount), 0) AS total_pledged
FROM campaigns c
LEFT JOIN pledges p ON c.campaign_id = p.campaign_id
GROUP BY c.campaign_id, c.title, c.goal_amount;
